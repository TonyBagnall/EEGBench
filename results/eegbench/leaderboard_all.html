Reproducing this page

from eegbench.datasets import eegbench
from eegbench.experiments.tables import leaderboard

leaderboard(
    datasets=sorted(eegbench),
    estimators=["HC2", "STC", "DrCIF", "MiniROCKET", "QUANT", "Arsenal", "ROCKET", "FreshPRINCE", "MRHydra", "TDE", "SVM", "EEGNet", "DCN", "LogisticRegression", "IT", "RandomForest", "CNN", "R-KNN", "CSP-SVM", "Default", "R-MDM"],
    metrics=["accuracy", "balacc", "auroc", "f1", "logloss", "sensitivity", "specificity"],
    sort_by="accuracy",
)

Or python -m eegbench.experiments.tables to rebuild it with the defaults.
